\paper{数学类}

\begin{problems}

\item[一、]（本题 15 分）设 $S$ 是空间中的一个椭球面．设方向为常向量 $V$ 的一束平行光照射 $S$，其中部分光线与 $S$ 相切，它们的切点在 $S$ 上形成一条曲线 $\Gamma$．证明：$\Gamma$ 落在一张过椭球中心的平面上．

\item[二、]（本题 15 分）设 $n$ 为奇数，$A,B$ 为两个实 $n$ 阶方阵，且 $BA=0$．记 $A+J_A$ 的特征值集合为 $S_1$，$B+J_B$ 的特征值集合为 $S_2$，其中 $J_A,J_B$ 分别表示 $A,B$ 的 Jordan 标准型．求证：$0\in S_1\cup S_2$．

\item[三、]（本题 20 分）设 $A_1,\cdots,A_{2017}$ 为 $2016$ 阶实方阵．证明关于 $x_1,\cdots,x_{2017}$ 的方程
\[
  \det(x_1A_1+\cdots+x_{2017}A_{2017})=0
\]
至少有一组非零实数解，其中 $\det$ 表示行列式．

\item[四、]（本题 20 分）设 $f_0(x),f_1(x)$ 是 $[0,1]$ 上正连续函数，满足 $\displaystyle\int_0^1 f_0(x)\dd x\le\int_0^1 f_1(x)\dd x$．

设
\[
  f_{n+1}(x)=\frac{2f_n^2(x)}{f_n(x)+f_{n-1}(x)},\qquad n=1,2,\cdots.
\]
求证：数列 $\displaystyle a_n=\int_0^1 f_n(x)\dd x$，$n=1,2,\cdots$ 单调递增且收敛．

\item[五、]（本题 15 分）设 $\alpha>1$．求证：不存在 $[0,+\infty)$ 上的正可导函数 $f(x)$ 满足
\begin{equation}
  f'(x)\ge f^{\alpha}(x),\qquad x\in[0,+\infty).\tag{1}
\end{equation}

\item[六、]（本题 15 分）设 $f(x),g(x)$ 是 $[0,1]$ 区间上的单调递增函数，满足
\[
  0\le f(x),g(x)\le 1,\qquad \int_0^1 f(x)\dd x=\int_0^1 g(x)\dd x.
\]
求证：$\displaystyle\int_0^1\abs{f(x)-g(x)}\dd x\le\frac{1}{2}$．

\end{problems}
