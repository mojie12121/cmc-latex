\paper{数学 A 类}

\begin{problems}

\item[一、]（本题 15 分）在空间直角坐标系中已知单叶双曲面 $S$ 的方程为 $x^2+y^2-z^2=1$。求过 $P=(1,1,1)$ 点落在单叶双曲面 $S$ 上的两条直线之间的夹角。

\item[二、]（本题 15 分）设 $\lim\limits_{n\to+\infty}\frac{a_n}{n^2}=a$，$\lim\limits_{n\to+\infty}\frac{b_n}{n^2}=b$。证明极限
$\lim\limits_{n\to+\infty}\frac{1}{n^5}\sum\limits_{k=0}^{n}a_kb_{n-k}$
存在并求其值。

\item[三、]（本题 15 分）设 $A=\begin{pmatrix}2&1\\1&1\end{pmatrix}$，矩阵 $B$ 与 $A$ 可交换，其元素均为正整数且行列式为 $1$。证明存在正整数 $k$ 使得 $B=A^k$。

\item[四、]（本题 20 分）设 $n\ge2$ 为正整数，证明多项式 $f(x)=x^n-x-1$ 在有理数域 $\Q$ 上不可约。

\item[五、]（本题 15 分）设 $\lim\limits_{n\to+\infty}\beta_n=0$，函数 $f$ 在 $[-1,2]$ 上有界，在 $[0,1]$ 上 Riemann 可积。证明：
\[
\lim_{n\to+\infty}\frac{1}{n}\sum_{k=1}^{n}f\left(\frac{k}{n}+\beta_n\right)=\int_0^1f(x)\dd x.
\]

\item[六、]（本题 20 分）设 $f$ 在 $[0,+\infty)$ 的任意闭区间上 Riemann 可积。对于 $x\ge0$，定义
\[
F(x)=\int_0^xt^\alpha f(t+x)\dd t.
\]
\begin{enumerate}
  \item 若 $\alpha\in(-1,0)$ 且 $\lim\limits_{x\to+\infty}f(x)=A$，证明：$F$ 在 $[0,+\infty)$ 上一致连续。
  \item 若 $\alpha\in(0,1)$，$f$ 以 $T>0$ 为周期，$\int_0^3f(t)\dd t=2022$。证明：$F$ 在 $[0,+\infty)$ 上非一致连续。
\end{enumerate}

\end{problems}
