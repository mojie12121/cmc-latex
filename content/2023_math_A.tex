\paper{数学 A 类}

\begin{problems}
\item[一、]（本题 15 分）在空间中给定直线 $L$ 及直线外定点 $P$．设 $M$ 是过 $P$ 点且与直线 $L$ 相切的球面的球心．问：所有可能的球心 $M$ 构成何种曲面？证明你的结论．

\item[二、]（本题 15 分）设 $f(x,y,z)=x^{2}+(y^{2}+z^{2})(1-x)^{3}$．
\begin{enumerate}[label=(\arabic*)]
  \item 计算 $f$ 的驻点．
  \item 求 $f$ 在 $\Sigma$ 上的最小值，其中，$\Sigma$ 是 $\{(x,y,z)\mid |x|\le 2,\ y^{2}+z^{2}\le 4\}$ 的边界．
  \item 求 $f$ 在椭球 $x^{2}+\frac{y^{2}}{2}+\frac{z^{2}}{3}\le 1$ 上的最小值．
\end{enumerate}

\item[三、]（本题 20 分）设 $V$ 是复数域 $\C$ 上的 $n$ 维线性空间，$\bm{A}$ 是 $V$ 上的一个线性变换．证明：存在 $\alpha\in V$ 使得 $\{\alpha,\bm{A}\alpha,\cdots,\bm{A}^{n-1}\alpha\}$ 成为 $V$ 的一组基当且仅当对于 $\bm{A}$ 的任一特征值 $\lambda$，$\lambda$ 的几何重数为 $1$．

\item[四、]（本题 15 分）设 $n\ge 3$ 为自然数，$\theta=\frac{2\pi}{n}$．对任意 $1\le s,t\le n$，取 $a_{st}=\sin\bigl((s+t)\theta\bigr)$，令矩阵 $\bm{A}=(a_{st})_{n\times n}$，计算 $\bm{E}+\bm{A}^{2023}$ 的行列式，其中 $\bm{E}$ 为 $n$ 阶单位矩阵．

\item[五、]（本题 15 分）设 $E\subset\R^{n}$ 非空有界，$\bm{c}\in\R^{n}$，$\bm{c}$ 非零．用
\[
  \operatorname{diam}E=\sup_{x,y\in E}\norm{\bm{x}-\bm{y}}
\]
表示 $E$ 的直径，记 $E+\bm{c}=\{\bm{x}+\bm{c}\mid \bm{x}\in E\}$．证明：$\operatorname{diam}E<\operatorname{diam}\bigl(E\cup(E+\bm{c})\bigr)$．

\item[六、]（本题 20 分）设 $a=\sqrt[3]{3}$，$x_{1}=a$，$x_{n+1}=a^{x_{n}}\ (n=1,2,\dots)$．证明：数列 $\{x_{n}\}_{n=1}^{\infty}$ 极限存在，但不是 $3$．
\end{problems}
