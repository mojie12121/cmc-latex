\paper{非数学类}

\begin{problems}

\item[一、]（本题 30 分，每小题 6 分）填空题：
\begin{enumerate}
  \item $\displaystyle\lim_{x\to+\infty}\sqrt{x^{2}+x+1}\cdot\frac{x-\ln(\ee^{x}+x)}{x}=\underline{\hspace{2.5cm}}$．
  \item 设 $z=z(x,y)$ 是由方程 $2\sin(x+2y-3z)=x+2y-3z$ 所确定的二元隐函数，则 $\dfrac{\partial z}{\partial x}+\dfrac{\partial z}{\partial y}=\underline{\hspace{2.5cm}}$．
  \item 设函数 $f(x)$ 连续，且 $f(0)\neq0$，则 $\displaystyle\lim_{x\to0}\frac{2\int_{0}^{x}(x-t)f(t)\dd t}{x\int_{0}^{x}f(x-t)\dd t}=\underline{\hspace{2.5cm}}$．
  \item 过三条直线
  \[
  L_{1}:\begin{cases}x=0,\\ y-z=2,\end{cases}\quad
  L_{2}:\begin{cases}x=0,\\ x+y-z+2=0,\end{cases}\quad\text{与}\quad
  L_{3}:\begin{cases}x=\sqrt{2},\\ y-z=0\end{cases}
  \]
  的圆柱面方程为 $\underline{\hspace{2.5cm}}$．
  \item 记 $D=\left\{(x,y)\mid x^{2}+y^{2}\le\pi\right\}$，则
  \[
  \iint_{D}\left(\sin x^{2}\cos y^{2}+x\sqrt{x^{2}+y^{2}}\right)\dd x\dd y=\underline{\hspace{2.5cm}}.
  \]
\end{enumerate}

\item[二、]（本题 14 分）设 $x_{1}=2021$，$x_{n}^{2}-2(x_{n}+1)x_{n+1}+2021=0$（$n\ge1$），证明数列 $\{x_{n}\}$ 收敛，并求极限 $\displaystyle\lim_{n\to\infty}x_{n}$．

\item[三、]（本题 14 分）设 $f(x)$ 在 $[0,+\infty)$ 上是有界连续函数，证明：方程 $y''+14y'+13y=f(x)$ 的每一个解在 $[0,+\infty)$ 上都是有界函数．

\item[四、]（本题 14 分）对于 4 次齐次函数
\[
f(x,y,z)=a_{1}x^{4}+a_{2}y^{4}+a_{3}z^{4}+3a_{4}x^{2}y^{2}+3a_{5}y^{2}z^{2}+3a_{6}x^{2}z^{2},
\]
计算曲面积分 $\displaystyle\iint_{\Sigma}f(x,y,z)\dd S$，其中 $\Sigma$：$x^{2}+y^{2}+z^{2}=1$．

\item[五、]（本题 14 分）设函数 $f(x)$ 在闭区间 $[a,b]$ 上有连续的二阶导数，证明：
\[
\lim_{n\to\infty}n^{2}\left[\int_{a}^{b}f(x)\dd x-\frac{b-a}{n}\sum_{k=1}^{n}f\left(a+\frac{2k-1}{2n}(b-a)\right)\right]=\frac{(b-a)^{2}}{24}\left[f'(b)-f'(a)\right].
\]

\item[六、]（本题 14 分）设 $\{a_{n}\}$ 与 $\{b_{n}\}$ 均为正实数列，满足：$a_{1}=b_{1}=1$，且 $b_{n}=a_{n}b_{n-1}-2$，$n=2,3,\cdots$．又设 $\{b_{n}\}$ 为有界数列，证明级数 $\displaystyle\sum_{n=1}^{\infty}\frac{1}{a_{1}a_{2}\cdots a_{n}}$ 收敛，并求该级数的和．

\end{problems}
