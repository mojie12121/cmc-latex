\paper{数学 B 类}

\begin{problems}
\item[一、]（本题 15 分）在空间中给定两不同点 $P$ 和 $Q$．过 $P$ 点直线 $L(P)$ 和过 $Q$ 点直线 $L(Q)$ 正交于点 $M$．问：所有可能的正交点 $M$ 构成何种曲面？证明你的结论．

\item[二、]（本题 15 分）设 $f(x,y,z)=x^{2}+(y^{2}+z^{2})(1-x)^{3}$．
\begin{enumerate}[label=(\arabic*)]
  \item 计算 $f$ 的驻点．
  \item 求 $f$ 在 $\Sigma$ 上的最小值，其中，$\Sigma$ 是 $\{(x,y,z)\mid |x|\le 2,\ y^{2}+z^{2}\le 4\}$ 的边界．
  \item 求 $f$ 在椭球 $x^{2}+\frac{y^{2}}{2}+\frac{z^{2}}{3}\le 1$ 上的最小值．
\end{enumerate}

\item[三、]（本题 20 分）设 $V$ 是复数域 $\C$ 上的 $n$ 维线性空间，$\bm{A}$ 是 $V$ 上的一个线性变换．证明：存在 $\alpha\in V$ 使得 $\{\alpha,\bm{A}\alpha,\cdots,\bm{A}^{n-1}\alpha\}$ 成为 $V$ 的一组基当且仅当对于 $\bm{A}$ 的任一特征值 $\lambda$，$\lambda$ 的几何重数为 $1$．

\item[四、]（本题 15 分）证明对任意 $n$ 阶方阵 $A$，存在主对角线上元素为 $1$ 或 $-1$ 的 $n$ 阶对角矩阵 $J$ 使得 $A+J$ 可逆．

\item[五、]（本题 15 分）设 $f(x)=x^{n}(1-x)^{n}$，
\[
  F(x)=f(x)-f''(x)+f^{(4)}(x)-\cdots+(-1)^{n}f^{(2n)}(x).
\]
计算并化简 $\dfrac{\mathrm{d}}{\mathrm{d}x}\bigl(F'(x)\sin x-F(x)\cos x\bigr)$．

\item[六、]（本题 20 分）设非负函数 $f$ 在 $[0,+\infty)$ 上连续可微，无穷积分 $\displaystyle\int_{0}^{+\infty}f(x)\dd x$ 收敛，且存在 $[0,+\infty)$ 上的非负函数 $g$，使得
\[
  f'(x)\le g(x),\qquad x\ge 0. \tag{1}
\]
分别就下列三种情形，证明 $\displaystyle\lim_{x\to+\infty}f(x)=0$．
\begin{enumerate}[label=(\roman*)]
  \item $\displaystyle\int_{0}^{+\infty}g(x)\dd x$ 收敛．
  \item $g(x)=C>0$，其中 $C$ 为常数．
  \item $g(x)=Cf^{p}(x)$，其中 $C>0$，$p>0$ 为常数．
\end{enumerate}
\end{problems}
