\paper{非数学 A 类}

\begin{problems}
\item[一、]（本题 30 分，每小题 6 分）填空题：
\begin{enumerate}
\item $\displaystyle\int_0^1 \ln(1+x^2)\dd x=$ \underline{\hspace{6em}}.
\item 设 $D: x^2+y^2\le r^2$，其中 $r>0$. 则
\[ \lim_{r\to 0^+}\frac{\iint_D (\mathrm{e}^{x^2+y^2}-1)\dd x\dd y}{r^4}=\underline{\hspace{6em}}. \]
\item 已知函数 $z=f(xy,\mathrm{e}^{x+y})$，且 $f(x,y)$ 具有二阶连续偏导数. 则
\[ \frac{\partial^2 z}{\partial x\partial y}=\underline{\hspace{10em}}. \]
\item 直线 $L:\dfrac{x-1}{1}=\dfrac{y}{1}=\dfrac{z-1}{-1}$ 在平面 $\pi: x-y+2z-1=0$ 上的投影直线 $L_0$ 的单位方向向量为 \underline{\hspace{6em}}.
\item 设 $L$ 为圆周 $x^2+y^2=9$，取逆时针方向，则第二型曲线积分
\[ \int_L \frac{-y}{4x^2+y^2}\dd x+\frac{x}{4x^2+y^2}\dd y=\underline{\hspace{6em}}. \]
\end{enumerate}
\item[二、]（本题 14 分）求微分方程 $(x^3-y^2)\dd x+(x^2y+xy)\dd y=0$ 的通解.
\item[三、]（本题 14 分）设函数
\[ f(x)=\begin{cases}\dfrac{1}{\ln(1+x)}-\dfrac{1}{x}, & 0<x\le 1,\\[6pt] k, & x=0.\end{cases} \]
\begin{enumerate}
\item 求常数 $k$ 的值，使得 $f(x)$ 在区间 $[0,1]$ 上连续；
\item 对 (1) 中 $k$ 的值，求函数 $f(x)$ 的最小值 $\lambda$ 与最大值 $\mu$.
\end{enumerate}
\item[四、]（本题 14 分）求曲面积分
\[ I=\iint_S (x^2-x)\dd y\dd z+(y^2-y)\dd z\dd x+(z^2-z)\dd x\dd y, \]
其中 $S$ 是上半球面 $x^2+y^2+z^2=R^2$（$z\ge 0$）的上侧.
\item[五、]（本题 14 分）设 $f(x)$ 是 $(-\infty,+\infty)$ 上具有连续导数的非负函数，且存在 $M>0$ 使得对任意的 $x,y\in(-\infty,+\infty)$，有 $|f'(x)-f'(y)|\le M|x-y|$. 证明：对于任意实数 $x$，恒有 $(f'(x))^2\le 2Mf(x)$.
\item[六、]（本题 14 分）证明：级数 $\displaystyle\sum_{n=1}^{\infty}\sum_{k=1}^{\infty}\frac{(-1)^{[\sqrt{n}]}}{n^2+k^2}$ 收敛，其中 $[x]$ 表示不超过 $x$ 的最大整数.
\end{problems}
